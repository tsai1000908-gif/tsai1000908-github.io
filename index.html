import random # 召喚處理「隨機」的魔法模組 # ========================================== # Python 猜拳遊戲（5大道具豪華版） # ========================================== # 1. 冒險者登入 name = input("請輸入你的冒險者名字：") equipment = input("請選擇你的使用裝備（例如：絞鯊狂鱗、魔導神杖）：") # 印出前置劇情 print("\n=== 系統廣播 ===") print(f"歡迎，新的冒險者【{name}】！") print(f"你已經裝備了【{equipment}】，準備好迎接第一關的 【猜拳】 挑戰！") # 🔄 最外層迴圈：用來控制「重新開始遊戲」 while True: print("\n\n=== ✊✋✌ 剪刀石頭布 終極大對決 ===") print("規則：先獲得 3 勝的一方就是最終贏家！\n") # 戰鬥前置作業：每次新遊戲重置數據與背包 player_wins = 0 computer_wins = 0 round_num = 1 # 🎒 背包道具數量 mirror_count = 2 # 偷窺之鏡 shield_count = 1 # 聖光護盾 clover_count = 1 # 幸運四葉草 double_count = 1 # 雙倍卡 curse_count = 1 # 混亂術 options = ["剪刀", "石頭", "布"] # 2. 遊戲主迴圈 while player_wins < 3 and computer_wins < 3: print(f"--- 第 {round_num} 回合 ---") print(f"🏆 目前戰況 - 你：{player_wins} 勝 | 電腦：{computer_wins} 勝") print(f"🎒 背包道具：") print(f" 🔮偷窺之鏡：{mirror_count}次 | 🛡️聖光護盾：{shield_count}次 | 🍀幸運四葉草：{clover_count}次") print(f" 💥雙倍卡：{double_count}次 | 🌀混亂術：{curse_count}次") # 本回合狀態標記 use_shield_this_round = False use_clover_this_round = False score_multiplier = 1 # 預設勝負算 1 分 # 決定電腦這回合要出什麼（預設可選全招式） available_options = options.copy() # 🌀 道具 1：混亂術（封印電腦其中一種出拳） if curse_count > 0: use_curse = input("\n要使用「🌀混亂術」嗎？(輸入 use 隨機封印電腦一招，其他鍵不用)：").lower() if use_curse == "use": curse_count -= 1 banned_option = random.choice(available_options) available_options.remove(banned_option) print(f"🌀【混亂術發動！】成功封印電腦！這回合電腦絕對不會出【{banned_option}】！") # 電腦在剩餘可用的招式中抽出一種 computer_choice = random.choice(available_options) # 🔮 道具 2：偷窺之鏡 if mirror_count > 0: use_mirror = input("要使用「🔮偷窺之鏡」嗎？(輸入 use 使用，其他鍵不用)：").lower() if use_mirror == "use": mirror_count -= 1 if random.random() < 0.5: print(f"✨【偷窺之鏡發動成功！】透過魔法鏡面，你預知電腦準備出【{computer_choice}】！") else: print("💨【偷窺之鏡被發現了...】發動失敗，什麼都沒看到！") # 🛡️ 道具 3：聖光護盾 if shield_count > 0: use_s = input("要使用「🛡️聖光護盾」嗎？(輸入 use 使用，輸了不扣分，其他鍵不用)：").lower() if use_s == "use": shield_count -= 1 use_shield_this_round = True print("🛡️【聖光護盾展開！】這回合如果輸了將不計分數。") # 🍀 道具 4：幸運四葉草 if clover_count > 0: use_c = input("要使用「🍀幸運四葉草」嗎？(輸入 use 使用，平手直接算勝，其他鍵不用)：").lower() if use_c == "use": clover_count -= 1 use_clover_this_round = True print("🍀【四葉草祝福！】這回合如果平手將直接視為你獲勝！") # 💥 道具 5：雙倍卡 if double_count > 0: use_d = input("要使用「💥雙倍卡」嗎？(輸入 use 使用，本回合分數加倍，其他鍵不用)：").lower() if use_d == "use": double_count -= 1 score_multiplier = 2 print("💥【雙倍下注！】這回合贏家將一次獲得 2 分！") # 3. 玩家出拳 player_choice = input("\n請出拳 (輸入 剪刀、石頭 或 布)：") # 檢查輸入錯誤 if player_choice not in options: print("⚠️ 吼！指令錯誤，請輸入 剪刀、石頭 或 布。\n") continue # 4. 印出電腦出的拳 print(f"🤖 天才電腦出了：{computer_choice}") # 5. 勝負判斷 if player_choice == computer_choice: if use_clover_this_round: points = 1 * score_multiplier print(f"🍀 雖然平手，但在四葉草的祝福下，算你獲得 {points} 勝！\n") player_wins += points else: print("...居然都出一樣，沒有人得分。\n") elif (player_choice == "剪刀" and computer_choice == "布") or \ (player_choice == "石頭" and computer_choice == "剪刀") or \ (player_choice == "布" and computer_choice == "石頭"): points = 1 * score_multiplier print(f"🎉 這回合你贏了！獲得 {points} 分！\n") player_wins += points else: if use_shield_this_round: print("🛡️ 這回合你輸了，但聖光護盾抵擋了傷害，電腦無法得分！\n") else: points = 1 * score_multiplier print(f"💥 這回合電腦贏了！獲得 {points} 分！\n") computer_wins += points round_num += 1 # 6. 最終結算 print("=== 🌟 最終結果 🌟 ===") if player_wins >= 3: print(f"🏆 恭喜你以 {player_wins} : {computer_wins} 擊敗了天才電腦，成為猜拳大師！") else: print(f"💻 很可惜，電腦以 {computer_wins} : {player_wins} 獲得了最終勝利！") # 🔁 再來一場 again = input("\n想要再挑戰一場嗎？(輸入 yes 重新開始，輸入其他鍵離開)：").lower() if again != "yes": print(f"感謝遊玩，再見了【{name}】！") break
